# =====================================================================
# Shanfix BMS — serve the system from the domain root.
#
# Upload this file as ".htaccess" into the folder your subdomain points
# at — the one containing app/, config/, public/, database/.
#
# It forwards every request into public/ so visitors use
#   https://system.shanfixtechnology.com/login
# instead of
#   https://system.shanfixtechnology.com/public/login
#
# IMPORTANT: also add  'base_path' => '',  to the 'app' block in
# config/config.php, or generated links will still say /public/.
# =====================================================================

Options -Indexes

<IfModule mod_rewrite.c>
    RewriteEngine On

    # --- Never serve project internals -------------------------------
    # These sit inside the web root under this layout, so they are
    # blocked explicitly as well as by their own .htaccess files.
    RewriteRule ^(app|config|database|storage)(/|$)          - [F,L]
    RewriteRule ^(install|migrate|cron|dev-server|routes)\.php$ - [F,L]
    RewriteRule ^\.                                          - [F,L]
    RewriteRule \.(sql|log|ini|sh|bak|md|yml|yaml)$           - [F,L]

    # --- Forward everything else into public/ ------------------------
    # The (?!public/) lookahead is what stops this looping: after the
    # first rewrite the path already starts with public/, so the rule
    # no longer matches and mod_rewrite stops. Without it Apache loops
    # until it gives up with a 500.
    RewriteRule ^(?!public/)(.*)$ public/$1 [L]
</IfModule>

# Belt and braces for hosts where mod_rewrite is unavailable.
<FilesMatch "\.(sql|log|ini|sh|bak|md|yml|yaml)$">
    Require all denied
</FilesMatch>

<FilesMatch "^\.">
    Require all denied
</FilesMatch>
