# =====================================================================
# FALLBACK ONLY — use this if you cannot change the subdomain's
# document root in cPanel.
#
# The correct fix is cPanel -> Domains -> set the document root for
# system.shanfixtechnology.com to:
#     /home/YOURUSER/system.shanfixtechnology.com/public
#
# Only if that is not possible: rename this file to ".htaccess" and put
# it in the FOLDER THE SUBDOMAIN CURRENTLY POINTS AT (the one holding
# app/, config/, public/ …). It forwards every request into public/.
#
# Note the trade-off: with this approach your project files stay inside
# the web root, so they are protected only by the .htaccess deny rules,
# not by being physically unreachable.
# =====================================================================

<IfModule mod_rewrite.c>
    RewriteEngine On

    # Force HTTPS
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !https
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

    # Never expose the project internals, whatever else happens
    RewriteRule ^(app|config|database|storage)(/|$) - [F,L]
    RewriteRule ^(install|migrate|cron|dev-server|routes)\.php$ - [F,L]

    # Send everything else into public/
    RewriteCond %{REQUEST_URI} !^/public/
    RewriteRule ^(.*)$ public/$1 [L]
</IfModule>

Options -Indexes

# Block files that should never be downloadable
<FilesMatch "\.(sql|log|ini|sh|bak|md|yml|yaml|lock)$">
    Require all denied
</FilesMatch>

<FilesMatch "^\.">
    Require all denied
</FilesMatch>
